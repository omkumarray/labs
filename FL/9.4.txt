void main() {
  // Enter your number here
  int number = 5;

  print("Number: $number");

  // Simple if statement
  if (number > 0) {
    print("The number is positive.");

    if (number % 2 == 0) {
      print("The number is even.");
    } else {
      print("The number is odd.");
    }
  } else if (number < 0) {
    print("Negative number");
  } else {
    print("Zero");
  }
}
